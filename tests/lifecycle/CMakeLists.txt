cmake_minimum_required(VERSION 3.16)
project(PlayerbotLifecycleMathTests LANGUAGES CXX)
enable_testing()
add_executable(lifecycle_math
    main.cpp
    ../../playerbot/RandomBotLifecycleMath.cpp
    ../../playerbot/strategy/tests/RandomBotLifecycleMathTest.cpp)
target_compile_features(lifecycle_math PRIVATE cxx_std_14)
target_compile_definitions(lifecycle_math PRIVATE PLAYERBOT_LIFECYCLE_STANDALONE_TEST)
target_include_directories(lifecycle_math PRIVATE ../..)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(lifecycle_math PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME lifecycle_math COMMAND lifecycle_math)
